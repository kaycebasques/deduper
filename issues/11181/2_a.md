If anyone finds this, I was able to work around this limitation with a couple of event hooks.

```python
# Mangled names to be skipped.
mangled_names = set()

# For every name-mangled attribute in the class, create an attribute with its
# original, unmangled name (and the same value).  If necessary, also create an
# attribute with the name mangled as Sphinx expects, if it is different than the
# actual mangled name.  (See https://github.com/sphinx-doc/sphinx/pull/14554.)
def alias_mangled_attributes(app, what, name, obj, options, lines):
    if what == 'module':
        mangled_names.clear()
    elif what in ('class', 'exception') and isinstance(obj, type):
        runtime_prefix = f'_{obj.__name__.lstrip('_')}__'
        for runtime_name, value in list(getattr(obj, '__dict__', {}).items()):
            if runtime_name.startswith(runtime_prefix):
                orig_name = f'__{runtime_name.removeprefix(runtime_prefix)}'
                sphinx_name = f'_{obj.__name__}{orig_name}'
                setattr(obj, orig_name, value)
                mangled_names.add(runtime_name)
                if sphinx_name != runtime_name:
                    setattr(obj, sphinx_name, value)
                    mangled_names.add(sphinx_name)

# Skip attributes with mangled names
def skip_mangled_members(app, what, name, obj, skip, options):
    if what in ('class', 'exception') and name in mangled_names:
            return True
    return skip

def setup(app):
    app.connect('autodoc-process-docstring' alias_mangled_attributes)
    app.connect('autodoc-skip-member', skip_mangled_members)
```

This causes Sphinx to document all "private" members or any sort using their unmangled names.
