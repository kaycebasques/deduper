Note: The error will be gone when I enabled imgconverter extension:
```
FROM python:3.8-slim

RUN apt update; apt install -y build-essential curl git make unzip vim
RUN git clone https://github.com/shuveb/loti
WORKDIR /loti
RUN pip install Sphinx
RUN apt install -y imagemagick
RUN echo "extensions.append('sphinx.ext.imgconverter')" >> conf.py
RUN make latex
```
