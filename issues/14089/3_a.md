In a couple of projects, where I'm still using Sphinx v7, I was able to override the default `AttributeDocumenter` with a custom one, that recognizes class attributes initialized with a  `SQL` literals (more specifically, [psycopg2 SQL instances](https://www.psycopg.org/docs/sql.html#psycopg2.sql.Composable)), that it renders as a `.. code-block:: postgresql`:

``` python
    from sphinx.ext import autodoc

    class AttributeDocumenter(autodoc.AttributeDocumenter):
        """Customized AttributeDocumenter that knows about ``SQL`` literals."""

        def add_directive_header(self, sig):
            """Omit the raw value in case of SQL instances."""
            if not isinstance(self.object, Psycopg2SQL.Composable):
                autodoc.AttributeDocumenter.add_directive_header(self, sig)
            else:
                autodoc.ClassLevelDocumenter.add_directive_header(self, sig)

        def add_content(self, more_content):
            """Prettify SQL statements."""
            autodoc.AttributeDocumenter.add_content(self, more_content)
            if isinstance(self.object, Psycopg2SQL.Composable):
                sql = self.object.as_string(None)
                self.add_line('', '')
                self.add_line('.. code-block:: postgresql', '')
                self.add_line('', '')
                for line in sql.splitlines():
                    self.add_line('   ' + line, '')

    def setup(app):
        """Setup the Sphinx environment."""
        from docutils.parsers.rst import directives

        app.add_autodocumenter(AttributeDocumenter)
```
