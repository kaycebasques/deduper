In Sphinx 4.1, Sphinx added `nitpick_ignore_regex` which helps in ignoring some undefined references. However, this might not be sufficient in some cases, or it requires a huge list of regular expressions in `conf.py`, making that file much less readable. 

In my own project, I added a more general hook based on a predicate function instead in order to suppress the nitpick warning and suggest to do it directly in Sphinx. The way I implemented is by creating an event handler for `warn-missing-reference` as follows:

```python
__all__ = ['nitpicky_function']

import sys
from typing import TYPE_CHECKING

if TYPE_CHECKING:
    from typing import Literal
 
    from sphinx.addnodes import pending_xref
    from sphinx.application import Sphinx
    from sphinx.config import Config
    from sphinx.domains import Domain

_nitpicky_functions = []

def nitpicky_function(func):
    """A decorator adding a new nitpicky predicate."""
    _nitpicky_functions.append(func)
    return func

def suppress_missing_reference_warning(app, domain, node):
    # type: (Sphinx, Domain | None, pending_xref) -> Literal[True, None]
    if not app.config.nitpick_use_filter_predicate:
        return None

    reftype, reftarget = node['reftype'], node['reftarget']
    for predicate in _nitpicky_functions:
        if predicate(reftype, reftarget, domain=domain, node=node):
            return True

    return None

###############################################################################

def setup(app):
    app.add_config_value('nitpick_use_filter_predicate', True, True)
    app.connect('warn-missing-reference', suppress_missing_reference_warning)
    return {'parallel_read_safe': True, 'parallel_write_safe': True}
```
