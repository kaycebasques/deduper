IMO, a Sphinx extension can easily implement this function; there is no need to implement it in the core.

```py
from docutils import nodes
from sphinx.application import Sphinx
from sphinx.util import logging
from sphinx.util.docutils import SphinxDirective

class lang_node(nodes.Inline, nodes.TextElement): ...

def html_visit_lang_node(self, node: lang_node) -> None:
    self.body.append(
        self.starttag(
            node,
            'div',
            '',
            LANG=node['lang'],
        )
    )

def html_depart_lang_node(self, node: lang_node) -> None:
    self.body.append('</div>')

class LangDirective(SphinxDirective):
    required_arguments = 1
    has_content = True

    def run(self) -> list[nodes.Node]:
        n = lang_node()
        n['lang'] = self.arguments[0]
        n += self.parse_content_to_nodes()
        return [n]


def setup(app: Sphinx):
    app.add_directive('lang', LangDirective)
    app.add_node(
        lang_node,
        html=(html_visit_lang_node, html_depart_lang_node),
    )
```

Write rst:
```rst
.. lang:: zh-CN

   本段落以中文纂写。
```

Which generates:
```html
<div lang="zh-CN"><p>本段落以中文纂写。</p>
</div>
```
