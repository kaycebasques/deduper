```python
from os import PathLike
import os
from pathlib import Path
from typing import Callable
from sphinx.application import Sphinx
from sphinx.environment import BuildEnvironment
from sphinx.util import logging, osutil

logger = logging.getLogger(__name__)

def filepath_to_docname(filepath: str) -> str:
    return str(Path(filepath).with_suffix(""))

def register_virtual_doc(app: Sphinx, filepath: str, docname: str, removed: set[str]):
    if (app.srcdir / filepath).exists():
        logger.warning(f"Virtual document '{filepath}' shadows real file")
    app.project.docnames.add(docname)
    removed.discard(docname)

injected_documents: dict[str, str] = {}
lazily_injected_documents: dict[str, str] = {}
lazy_injectors: list[Callable[[Sphinx, BuildEnvironment], tuple[str, str]]] = []
warnings_added: list[str] = []
warnings_to_suppress = ("toc", "ref", "myst")
build_phase = 1

_app: Sphinx | None = None
_doc2path_original: Callable[[str, bool], PathLike[str]] | None = None
_copyfile_original: Callable[[str | os.PathLike[str], str | os.PathLike[str]], None] = (
    osutil.copyfile
)

class VirtualPath(PathLike[str]):
    def __init__(self, path: Path, filename: str):
        self._path = Path(path)
        self._filename = filename

    def read_text(self, encoding: str | None = None, errors: str | None = None):
        if self._filename in lazily_injected_documents:
            return lazily_injected_documents[self._filename]
        return injected_documents[self._filename]

    def __fspath__(self):
        return str(self._path)

    def __str__(self):
        return str(self._path)

    def __getattr__(self, name: str):
        return getattr(self._path, name)

def patched_copyfile(
    source: str | os.PathLike[str], dest: str | os.PathLike[str], *, force: bool = False
):
    if _app:
        try:
            rel_path_str = str(Path(source).relative_to(_app.srcdir))
            for injections in (injected_documents, lazily_injected_documents):
                if rel_path_str in injections:
                    dest_path = Path(dest)
                    dest_path.parent.mkdir(parents=True, exist_ok=True)
                    dest_path.write_text(injections[rel_path_str], encoding="utf-8")
                    return
        except ValueError:
            pass
    _copyfile_original(source, dest, force=force)  # type: ignore

osutil.copyfile = patched_copyfile

def patched_doc2path(docname: str, absolute: bool = True) -> PathLike[str]:
    for filepath in injected_documents:
        if filepath_to_docname(filepath) == docname:
            path = (
                _app.project.srcdir / filepath
                if (absolute and _app)
                else Path(filepath)
            )
            return VirtualPath(path, filepath)
    if build_phase == 2:
        for filepath in lazily_injected_documents:
            if filepath_to_docname(filepath) == docname:
                path = (
                    _app.project.srcdir / filepath
                    if (absolute and _app)
                    else Path(filepath)
                )
                return VirtualPath(path, filepath)
    if _doc2path_original is None:
        raise RuntimeError("_original_doc2path is None")
    return _doc2path_original(docname, absolute)

def builder_inited(app: Sphinx):
    global _doc2path_original, _app
    _app = app
    _doc2path_original = app.project.doc2path
    app.project.doc2path = patched_doc2path  # type: ignore
    if build_phase == 1 and lazy_injectors:
        user_suppressions = getattr(app.config, "suppress_warnings", [])
        new_warnings = [w for w in warnings_to_suppress if w not in user_suppressions]
        user_suppressions.extend(new_warnings)
        warnings_added.extend(new_warnings)

def env_get_outdated(
    app: Sphinx,
    env: BuildEnvironment,
    added: set[str],
    changed: set[str],
    removed: set[str],
) -> list[str]:
    global build_phase
    for filepath in injected_documents:
        docname = filepath_to_docname(filepath)
        register_virtual_doc(app, filepath, docname, removed)
        if build_phase == 1 and docname in env.found_docs:
            added.add(docname)
    if build_phase == 2:
        late_docnames: list[str] = []
        for lazy_injector in lazy_injectors:
            try:
                filepath, content = lazy_injector(app, env)
            except Exception as e:
                logger.error(f"Lazy injector failed: {e}")
                raise
            if filepath in lazily_injected_documents:
                logger.warning(f"Virtual document '{filepath}' already injected and thus gets replaced!")
            lazily_injected_documents[filepath] = content
            docname = filepath_to_docname(filepath)
            register_virtual_doc(app, filepath, docname, removed)
            late_docnames.append(docname)
        changed.update(env.reread_always & env.found_docs)
        return late_docnames
    return []

def env_before_read_docs(app: Sphinx, env: BuildEnvironment, _docnames: list[str]):
    global build_phase
    if build_phase == 2:
        env._pickled_doctree_cache.clear()  # type: ignore[attr-defined]
        env._write_doc_doctree_cache.clear()  # type: ignore[attr-defined]
        if "master_doctree" in env.__dict__:
            del env.__dict__["master_doctree"]  # i think alabaster needs it

def build_finished(app: Sphinx, exception: Exception):
    global build_phase
    if exception:
        build_phase = 1
        return
    if build_phase == 1 and lazy_injectors:
        build_phase = 2
        suppressions = getattr(app.config, "suppress_warnings", [])
        suppressions[:] = [w for w in suppressions if w not in warnings_added]
        warnings_added.clear()
        app.build(force_all=True)
    if build_phase == 2:
        build_phase = 1

def setup(app: Sphinx):
    app.connect("builder-inited", builder_inited)
    app.connect("env-get-outdated", env_get_outdated)
    app.connect("env-before-read-docs", env_before_read_docs)
    app.connect("build-finished", build_finished)

def inject(filepath: str, content: str):
    if filepath in injected_documents:
        logger.warning(f"Virtual document '{filepath}' already injected and thus gets replaced!")
    injected_documents[filepath] = content

def register_lazy_injector(
    generator: Callable[[Sphinx, BuildEnvironment], tuple[str, str]],
):
    lazy_injectors.append(generator)

def lazy_injector(
    func: Callable[[Sphinx, BuildEnvironment], tuple[str, str]],
) -> Callable[[Sphinx, BuildEnvironment], tuple[str, str]]:
    register_lazy_injector(func)
    return func

lazy_injection = lazy_injector
inject_lazily = lazy_injector

__all__ = ["inject", "register_lazy_injector", "lazy_injector", "lazy_injection", "inject_lazily"]
```

Note that this is quick and dirty, hence weird things such as global state. It works but has a few issues: it has to patch sphinx internals, it does two builds and clears the cache completely between them, it hides warnings between them.

However, it shows what API I wish sphinx had so I wouldn't have to patch and use internal stuff.
