e
        if foo_value not in res:
            res[foo_value] = []
        res[foo_value].append(doc)
    most_common = max(res.items(), key=lambda x: len(x[1]))[0]
    content = ""
    for agg, docs in res.items():
        content += f"+ {agg}\n"
        for doc in docs:
            content += f"\t+ {doc}\n"
        content += "\n"
    return (
        f"sub2/{most_common}.rst",
        f"""
Foos
====

{content}
""",
```
